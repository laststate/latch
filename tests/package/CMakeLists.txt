cmake_minimum_required(VERSION 3.20)
project(latch-package-consumer LANGUAGES C CXX)
enable_testing()
find_package(laststate-latch 0.2 REQUIRED CONFIG)
add_executable(c-consumer main.c)
target_link_libraries(c-consumer PRIVATE laststate::latch)
add_test(NAME latch-package-c-consumer COMMAND c-consumer)
add_executable(cpp-consumer main.cpp)
target_link_libraries(cpp-consumer PRIVATE laststate::latch-cpp)
add_test(NAME latch-package-cpp-consumer COMMAND cpp-consumer)
add_executable(port-consumer port.c)
target_link_libraries(port-consumer PRIVATE laststate::latch-port-nordic)
add_test(NAME latch-package-port-consumer COMMAND port-consumer)
if(TARGET laststate::latch-linux)
  add_executable(linux-consumer linux.c)
  target_link_libraries(linux-consumer PRIVATE laststate::latch-linux)
  add_test(NAME latch-package-linux-consumer COMMAND linux-consumer)
endif()
